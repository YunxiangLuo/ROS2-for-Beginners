## 29.1 抓取姿态计算

### 29.1.1 抓取姿态分析

抓取姿态是指机械臂末端执行器在抓取目标物体时的位姿。一个完整的抓取姿态需要考虑以下几个方面：

抓取姿态的计算需要确定四个要素：接近方向（从哪个方向接近目标物体）、抓取角度（末端执行器相对于物体的姿态）、抓取高度（末端在物体上方的高度偏移）以及撤离方向（抓取后的撤离方向）。

抓取姿态计算的关键公式：

```
抓取位姿 = 物体位姿 × 抓取偏移变换
```

```
T_grasp = T_object × T_offset
```

其中 T_offset 是根据物体形状和夹爪类型确定的变换矩阵。

### 29.1.2 抓取姿态生成

```python
#!/usr/bin/env python3
"""抓取姿态计算示例"""
import math
from copy import deepcopy
from geometry_msgs.msg import Pose, PoseStamped
from tf_transformations import quaternion_from_euler, euler_from_quaternion
import numpy as np


class GraspPoseCalculator:
    """抓取姿态计算器"""

    @staticmethod
    def compute_grasp_pose(
        object_pose,
        approach_angle=0.0,
        approach_distance=0.1,
        grasp_height_offset=0.0
    ):
        """
        计算抓取位姿
        参数:
            object_pose: 物体位姿 (Pose)
            approach_angle: 接近角度 (弧度)
            approach_distance: 接近距离 (米)
            grasp_height_offset: 抓取高度偏移 (米)
        返回:
            (grasp_pose, approach_pose): 抓取位姿和接近预位姿
        """
        grasp_pose = deepcopy(object_pose)

        # 根据物体位姿调整抓取高度
        grasp_pose.position.z += grasp_height_offset

        # 计算接近预位姿（在抓取位姿上方一定距离）
        approach_pose = deepcopy(grasp_pose)
        approach_pose.position.z += approach_distance

        return grasp_pose, approach_pose

    @staticmethod
    def compute_top_grasp(object_pose, gripper_depth=0.05):
        """
        计算顶部抓取姿态（从上方抓取）
        适用场景：物体顶部平坦，可从上方直接抓取
        """
        grasp_pose = deepcopy(object_pose)

        # 顶部抓取：末端朝下，姿态由物体决定
        q = quaternion_from_euler(math.pi, 0, 0)
        grasp_pose.orientation.x = q[0]
        grasp_pose.orientation.y = q[1]
        grasp_pose.orientation.z = q[2]
        grasp_pose.orientation.w = q[3]

        # 调整高度到物体上表面
        grasp_pose.position.z += gripper_depth

        # 接近预位姿（上方10cm）
        approach_pose = deepcopy(grasp_pose)
        approach_pose.position.z += 0.1

        # 撤离位姿（再上升10cm）
        retreat_pose = deepcopy(grasp_pose)
        retreat_pose.position.z += 0.15

        return grasp_pose, approach_pose, retreat_pose

    @staticmethod
    def compute_side_grasp(object_pose, side_direction='x'):
        """
        计算侧面抓取姿态
        适用场景：物体侧面平整，需要从侧面夹取
        """
        grasp_pose = deepcopy(object_pose)

        if side_direction == 'x':
            # 从X方向抓取
            q = quaternion_from_euler(0, math.pi/2, 0)
        elif side_direction == 'y':
            # 从Y方向抓取
            q = quaternion_from_euler(math.pi/2, 0, 0)
        else:
            q = quaternion_from_euler(0, 0, 0)

        grasp_pose.orientation.x = q[0]
        grasp_pose.orientation.y = q[1]
        grasp_pose.orientation.z = q[2]
        grasp_pose.orientation.w = q[3]

        # 接近预位姿
        approach_pose = deepcopy(grasp_pose)
        if side_direction == 'x':
            approach_pose.position.x -= 0.1
        elif side_direction == 'y':
            approach_pose.position.y -= 0.1

        retreat_pose = deepcopy(approach_pose)

        return grasp_pose, approach_pose, retreat_pose

    @staticmethod
    def pose_to_transform(pose):
        """将Pose转换为4x4变换矩阵"""
        from tf_transformations import quaternion_matrix
        T = quaternion_matrix([
            pose.orientation.x,
            pose.orientation.y,
            pose.orientation.z,
            pose.orientation.w
        ])
        T[0, 3] = pose.position.x
        T[1, 3] = pose.position.y
        T[2, 3] = pose.position.z
        return T

    @staticmethod
    def transform_to_pose(T):
        """将4x4变换矩阵转换为Pose"""
        from tf_transformations import quaternion_from_matrix
        pose = Pose()
        pose.position.x = T[0, 3]
        pose.position.y = T[1, 3]
        pose.position.z = T[2, 3]
        q = quaternion_from_matrix(T)
        pose.orientation.x = q[0]
        pose.orientation.y = q[1]
        pose.orientation.z = q[2]
        pose.orientation.w = q[3]
        return pose


# 使用示例
def demo_grasp_calculation():
    # 假设物体位姿
    object_pose = Pose()
    object_pose.position.x = 0.4
    object_pose.position.y = 0.1
    object_pose.position.z = 0.15
    object_pose.orientation.w = 1.0

    # 计算顶部抓取
    calc = GraspPoseCalculator()
    grasp, approach, retreat = calc.compute_top_grasp(object_pose)

    print(f'抓取位姿: ({grasp.position.x:.3f}, {grasp.position.y:.3f}, {grasp.position.z:.3f})')
    print(f'接近位姿: ({approach.position.x:.3f}, {approach.position.y:.3f}, {approach.position.z:.3f})')
    print(f'撤离位姿: ({retreat.position.x:.3f}, {retreat.position.y:.3f}, {retreat.position.z:.3f})')
```

### 29.1.3 官方要点——抓取位姿生成的官方生态：GraspIt!、GPD 与 Deep Grasp

> 本节内容综合翻译自 MoveIt 2 官方文档（moveit.picknik.ai 的 Pick and Place 与 MoveIt Task Constructor 教程）、GraspIt!/GPD 项目官方文档与 PickNik Academy 培训课程，另参考 The Construct 的抓取课程与 Robotics Back-End 的 pick-and-place 实战教程。原文均为英文，此处为中文编译，供课后巩固与进阶阅读。

对于 29.1 节的抓取姿态计算，MoveIt 官方文档推荐了两条外部路线：其一是 GraspIt!（哥伦比亚大学仿真器），基于物体 mesh 与手掌模型枚举力闭合（force-closure）抓取并导出抓取分数；其二是 GPD（Grasp Pose Detection，东北大学开源），用 CNN 直接从点云预测候选抓取位姿并输出到 MoveIt 可用的位姿列表。MoveIt 官方的 Deep Grasp 文档把"感知 → 抓取候选 → IK 验证 → 执行"串成完整示例，官方还建议对所有候选位姿按"可达性 + 碰撞 + 接近方向可达"排序后再逐个尝试——这与练习第 1 题手动构造顶部/侧面抓取位姿的过程互为验证。

## 29.2 夹爪控制

### 29.2.1 夹爪控制方式

平行夹爪是最常用的末端执行器类型，通常由两个相对运动的夹指组成。

在MoveIt2中，夹爪通常作为一个独立的规划组进行控制：

```python
#!/usr/bin/env python3
"""夹爪控制示例"""
import time
import rclpy
from rclpy.node import Node
from moveit.planning import MoveItPy, PlanningComponent

class GripperController(Node):
    def __init__(self):
        super().__init__('gripper_controller')
        self.moveit = MoveItPy(node_name='moveit_py')
        self.gripper = PlanningComponent(
            self.moveit, 'gripper', 'gripper_link'
        )

    def open_gripper(self, width=0.65):
        """打开夹爪到指定宽度"""
        self.gripper.set_start_state_to_current_state()
        self.gripper.set_joint_value_target([width, width])
        plan = self.gripper.plan()
        if plan:
            self.gripper.execute(plan.trajectory)
            self.get_logger().info(f'夹爪打开: {width}')
            return True
        return False

    def close_gripper(self, width=0.0):
        """关闭夹爪到指定宽度"""
        self.gripper.set_start_state_to_current_state()
        self.gripper.set_joint_value_target([width, width])
        plan = self.gripper.plan()
        if plan:
            self.gripper.execute(plan.trajectory)
            self.get_logger().info(f'夹爪关闭: {width}')
            return True
        return False

    def grasp_object(self, object_width=0.04):
        """抓取物体：先打开到比物体略宽，然后闭合"""
        # 打开夹爪（比物体略宽）
        self.open_gripper(object_width + 0.01)
        time.sleep(0.5)

        # 闭合夹爪
        self.close_gripper(-0.01)
        time.sleep(0.5)

    def release_object(self):
        """释放物体：打开夹爪"""
        self.open_gripper(0.65)
        time.sleep(0.5)

    def gripper_sequence(self):
        """夹爪动作序列演示"""
        self.get_logger().info('开始夹爪动作序列')

        # 完全打开
        self.open_gripper(0.65)
        time.sleep(0.5)

        # 半开
        self.open_gripper(0.3)
        time.sleep(0.5)

        # 完全闭合
        self.close_gripper(0.0)
        time.sleep(0.5)

        # 模拟抓取和释放
        self.grasp_object(0.05)
        time.sleep(0.5)
        self.release_object()

        self.get_logger().info('夹爪动作序列完成')

def main(args=None):
    rclpy.init(args=args)
    node = GripperController()
    node.gripper_sequence()
    rclpy.shutdown()

if __name__ == '__main__':
    main()
```

```python
class AdvancedGripperControl(Node):
    def __init__(self):
        super().__init__('advanced_gripper_control')
        self.moveit = MoveItPy(node_name='moveit_py')
        self.gripper = PlanningComponent(
            self.moveit, 'gripper', 'gripper_link'
        )

    def controlled_grasp(self, max_effort=50.0):
        """带力矩控制的抓取"""
        # 设置速度
        self.gripper.set_max_velocity_scaling_factor(0.3)
        self.gripper.set_max_acceleration_scaling_factor(0.3)

        # 缓慢闭合夹爪
        self.gripper.set_start_state_to_current_state()
        self.gripper.set_joint_value_target([0.0, 0.0])
        plan = self.gripper.plan()
        if plan:
            # 分步执行，模拟力矩控制
            trajectory = plan.trajectory
            # 只执行前半段轨迹（缓慢接近）
            mid_point = len(trajectory.points) // 2
            trajectory.points = trajectory.points[:mid_point]
            self.moveit.execute(plan)
            self.get_logger().info('夹爪已接触物体')
            return True
        return False
```

> **执行说明**：moveit_py 的执行接口挂在 `MoveItPy` 对象上（`self.moveit.execute(plan)`），而非 `PlanningComponent`；上面按点数截断轨迹属概念示意，实际需构造 `PlanResult` 后交给 `execute`。执行依赖执行服务器在线（启动 `arm_only.launch.py` 方式2），课程统一封装 `course_lab_utils.moveit2` 中的 `ensure_execution_servers` 提供有界等待与明确报错。

## 29.3 Attach/Detach物体

### 29.3.1 物体附着

抓取物体后，需要将物体附着到末端执行器的碰撞模型中，使MoveIt2在规划时考虑物体的碰撞体积：

```python
from moveit_msgs.msg import AttachedCollisionObject, CollisionObject
from shape_msgs.msg import SolidPrimitive
from geometry_msgs.msg import Pose

class AttachManager:
    def __init__(self, planning_scene_monitor):
        self.psm = planning_scene_monitor

    def attach_to_end_effector(
        self,
        object_id,
        link_name='gripper_link',
        object_size=[0.04, 0.04, 0.08],
        touch_links=None
    ):
        """
        将物体附着到末端执行器
        参数:
            object_id: 物体ID
            link_name: 附着的连杆名称
            object_size: 物体尺寸 [x, y, z]
            touch_links: 允许触碰的连杆列表
        """
        if touch_links is None:
            touch_links = ['gripper_link', 'link5']

        # 创建碰撞物体
        co = CollisionObject()
        co.id = object_id
        co.header.frame_id = link_name
        co.operation = CollisionObject.ADD

        primitive = SolidPrimitive()
        primitive.type = SolidPrimitive.BOX
        primitive.dimensions = object_size

        pose = Pose()
        pose.position.z = object_size[2] / 2  # 物体中心在夹爪前方
        pose.orientation.w = 1.0

        co.primitives = [primitive]
        co.primitive_poses = [pose]

        # 创建附着物体
        aco = AttachedCollisionObject()
        aco.link_name = link_name
        aco.object = co
        aco.touch_links = touch_links

        self.psm.process_attached_collision_object(aco)
        print(f'物体 {object_id} 已附着到 {link_name}')

    def detach_from_end_effector(self, object_id, link_name='gripper_link'):
        """
        从末端执行器分离物体
        """
        co = CollisionObject()
        co.id = object_id
        co.operation = CollisionObject.REMOVE

        aco = AttachedCollisionObject()
        aco.link_name = link_name
        aco.object = co

        self.psm.process_attached_collision_object(aco)
        print(f'物体 {object_id} 已从 {link_name} 分离')

    def attach_cylinder(self, object_id, link_name, height, radius):
        """附着圆柱体物体"""
        co = CollisionObject()
        co.id = object_id
        co.header.frame_id = link_name
        co.operation = CollisionObject.ADD

        primitive = SolidPrimitive()
        primitive.type = SolidPrimitive.CYLINDER
        primitive.dimensions = [height, radius]

        pose = Pose()
        pose.position.z = height / 2
        pose.orientation.w = 1.0

        co.primitives = [primitive]
        co.primitive_poses = [pose]

        aco = AttachedCollisionObject()
        aco.link_name = link_name
        aco.object = co
        aco.touch_links = ['gripper_link', 'link5']

        self.psm.process_attached_collision_object(aco)
```

## 29.4 Pick-and-Place流程

### 29.4.1 完整抓取放置流程

一个标准的Pick-and-Place流程包含以下步骤：

首先移动到预抓取位姿，接近物体上方或前方；接着移动到抓取位姿，到达抓取位置；然后闭合夹爪抓取物体，并将物体附着加入规划场景；随后撤离，上升到安全高度。放置段与抓取段对称：移动到预放置位姿，接近放置位置上方；移动到放置位姿，到达放置位置；打开夹爪释放物体；从规划场景分离物体；最后撤离回到安全位置。

```python
#!/usr/bin/env python3
"""Pick-and-Place完整流程"""
import time
import math
from copy import deepcopy
import rclpy
from rclpy.node import Node
from geometry_msgs.msg import Pose, PoseStamped
from moveit_msgs.msg import CollisionObject, AttachedCollisionObject
from shape_msgs.msg import SolidPrimitive
from tf_transformations import quaternion_from_euler
from moveit.planning import MoveItPy, PlanningComponent

class PickAndPlace(Node):
    def __init__(self):
        super().__init__('pick_and_place')
        self.moveit = MoveItPy(node_name='moveit_py')
        self.psm = self.moveit.get_planning_scene_monitor()
        self.arm = PlanningComponent(
            self.moveit, 'arm_group', 'link5'
        )
        self.gripper = PlanningComponent(
            self.moveit, 'gripper', 'gripper_link'
        )

        # 配置规划参数
        self.arm.set_goal_position_tolerance(0.01)
        self.arm.set_goal_orientation_tolerance(0.02)
        self.arm.set_max_velocity_scaling_factor(0.5)
        self.arm.set_max_acceleration_scaling_factor(0.3)

    def go_home(self):
        """回到home位姿"""
        self.arm.set_named_target('home')
        plan = self.arm.plan()
        if plan:
            self.arm.execute(plan.trajectory)
            time.sleep(0.5)

    def setup_scene(self):
        """设置规划场景"""
        # 添加桌子
        co = CollisionObject()
        co.id = 'table'
        co.header.frame_id = 'base_link'
        co.operation = CollisionObject.ADD
        primitive = SolidPrimitive()
        primitive.type = SolidPrimitive.BOX
        primitive.dimensions = [0.8, 0.6, 0.02]
        pose = PoseStamped()
        pose.header.frame_id = 'base_link'
        pose.pose.position.z = -0.01
        pose.pose.orientation.w = 1.0
        co.primitives = [primitive]
        co.primitive_poses = [pose.pose]
        self.psm.process_collision_object(co)
        time.sleep(0.5)

    def move_to_pose(self, target_pose, desc=''):
        """规划并移动到目标位姿"""
        self.arm.set_start_state_to_current_state()
        self.arm.set_pose_target(target_pose.pose, 'link5')
        plan = self.arm.plan()
        if plan:
            self.get_logger().info(f'{desc}: 规划成功')
            self.arm.execute(plan.trajectory)
            return True
        self.get_logger().error(f'{desc}: 规划失败')
        return False

    def open_gripper(self):
        """打开夹爪"""
        self.gripper.set_start_state_to_current_state()
        self.gripper.set_joint_value_target([0.65, 0.65])
        plan = self.gripper.plan()
        if plan:
            self.gripper.execute(plan.trajectory)

    def close_gripper(self):
        """关闭夹爪"""
        self.gripper.set_start_state_to_current_state()
        self.gripper.set_joint_value_target([0.0, 0.0])
        plan = self.gripper.plan()
        if plan:
            self.gripper.execute(plan.trajectory)

    def create_pose(self, x, y, z, roll=0, pitch=0, yaw=0):
        """创建位姿"""
        pose = PoseStamped()
        pose.header.frame_id = 'base_link'
        pose.pose.position.x = x
        pose.pose.position.y = y
        pose.pose.position.z = z
        q = quaternion_from_euler(roll, pitch, yaw)
        pose.pose.orientation.x = q[0]
        pose.pose.orientation.y = q[1]
        pose.pose.orientation.z = q[2]
        pose.pose.orientation.w = q[3]
        return pose

    def pick(self, object_pose):
        """执行抓取"""
        # 1. 打开夹爪
        self.get_logger().info('步骤1: 打开夹爪')
        self.open_gripper()
        time.sleep(0.3)

        # 2. 移动到预抓取位姿（物体上方0.12m）
        pre_grasp_pose = deepcopy(object_pose)
        pre_grasp_pose.pose.position.z += 0.12
        self.get_logger().info('步骤2: 移动到预抓取位姿')
        if not self.move_to_pose(pre_grasp_pose, '预抓取'):
            return False
        time.sleep(0.5)

        # 3. 下降到抓取位姿
        self.get_logger().info('步骤3: 下降到抓取位姿')
        if not self.move_to_pose(object_pose, '抓取'):
            return False
        time.sleep(0.3)

        # 4. 闭合夹爪
        self.get_logger().info('步骤4: 闭合夹爪')
        self.close_gripper()
        time.sleep(0.5)

        # 5. 附着物体
        self.get_logger().info('步骤5: 附着物体到末端')
        self.attach_object('target_object')
        time.sleep(0.3)

        # 6. 撤离（上升0.1m）
        retreat_pose = deepcopy(object_pose)
        retreat_pose.pose.position.z += 0.1
        self.get_logger().info('步骤6: 撤离')
        if not self.move_to_pose(retreat_pose, '撤离'):
            return False
        time.sleep(0.3)

        return True

    def place(self, place_pose):
        """执行放置"""
        # 1. 移动到预放置位姿（放置点上方0.1m）
        pre_place_pose = deepcopy(place_pose)
        pre_place_pose.pose.position.z += 0.1
        self.get_logger().info('步骤7: 移动到预放置位姿')
        if not self.move_to_pose(pre_place_pose, '预放置'):
            return False
        time.sleep(0.5)

        # 2. 下降到放置位姿
        self.get_logger().info('步骤8: 下降到放置位姿')
        if not self.move_to_pose(place_pose, '放置'):
            return False
        time.sleep(0.3)

        # 3. 打开夹爪释放物体
        self.get_logger().info('步骤9: 打开夹爪释放物体')
        self.open_gripper()
        time.sleep(0.5)

        # 4. 分离物体
        self.get_logger().info('步骤10: 分离物体')
        self.detach_object('target_object')
        time.sleep(0.3)

        # 5. 撤离
        retreat_pose = deepcopy(place_pose)
        retreat_pose.pose.position.z += 0.1
        self.get_logger().info('步骤11: 撤离')
        if not self.move_to_pose(retreat_pose, '撤离'):
            return False
        time.sleep(0.3)

        return True

    def attach_object(self, object_id):
        """附着物体到末端"""
        co = CollisionObject()
        co.id = object_id
        co.header.frame_id = 'gripper_link'
        co.operation = CollisionObject.ADD

        primitive = SolidPrimitive()
        primitive.type = SolidPrimitive.BOX
        primitive.dimensions = [0.04, 0.04, 0.08]

        pose = Pose()
        pose.position.z = 0.04
        pose.orientation.w = 1.0

        co.primitives = [primitive]
        co.primitive_poses = [pose]

        aco = AttachedCollisionObject()
        aco.link_name = 'gripper_link'
        aco.object = co
        aco.touch_links = ['gripper_link', 'link5']

        self.psm.process_attached_collision_object(aco)

    def detach_object(self, object_id):
        """从末端分离物体"""
        co = CollisionObject()
        co.id = object_id
        co.operation = CollisionObject.REMOVE

        aco = AttachedCollisionObject()
        aco.link_name = 'gripper_link'
        aco.object = co

        self.psm.process_attached_collision_object(aco)

    def run(self):
        self.get_logger().info('开始Pick-and-Place演示')

        # 设置场景
        self.setup_scene()

        # 回到home
        self.go_home()

        # 定义抓取位置（桌子上方）
        grasp_pose = self.create_pose(0.3, 0.1, 0.15, math.pi, 0, 0)

        # 定义放置位置
        place_pose = self.create_pose(0.3, -0.2, 0.15, math.pi, 0, 0)

        # 执行抓取
        if self.pick(grasp_pose):
            self.get_logger().info('抓取成功')
        else:
            self.get_logger().error('抓取失败')
            return

        time.sleep(1)

        # 执行放置
        if self.place(place_pose):
            self.get_logger().info('放置成功')
        else:
            self.get_logger().error('放置失败')

        # 回到home
        self.go_home()
        self.get_logger().info('Pick-and-Place演示完成')

def main(args=None):
    rclpy.init(args=args)
    node = PickAndPlace()
    node.run()
    rclpy.shutdown()

if __name__ == '__main__':
    main()
```

### 29.4.3 官方要点——官方 Pick and Place 教程：三段式抓取状态机

MoveIt 官方教程把本章 29.4 节的十步流程压缩为官方三段式范式：pre-grasp approach（沿撤离方向反向接近）、grasp（夹爪闭合 + attach）、post-grasp retreat（撤离到安全高度）——放置段与之对称（pre-place、place、retreat）。官方实现中"attach/detach"并非透明魔法：`ModifyPlanningScene` service 在抓取瞬间把物体加入规划场景并随末端连杆运动，在释放瞬间移除附着，这正是本章 29.3 节代码的官方原型；夹爪则通过 `GripperCommand` action 控制，规划器全程只负责臂部轨迹。

### 29.4.4 官方要点——MoveIt Task Constructor：官方的"阶段化"流水线

本章的十步流程在官方更复杂的工程中由 MoveIt Task Constructor（MTC）承接：它把 pick-and-place 建模为串联的阶段（GenerateGraspPose → ComputeIK → ModifyPlanningScene → MoveTo → ...），每个阶段独立验证、可并行生成多个解，失败时按顺序回溯而不是整条流程重跑。官方 MTC 文档的 pick-place 示例与练习第 2、3 题的差异在于多物体调度：MTC 的 `SerialContainer` 可以逐物体实例化整条流水线，天然适合分拣与码垛；官方教程同时演示了用 `current_state` 阶段锚定 home 位姿，避免"回 home"写死关节值。

```python
class ConveyorBeltGrasp(Node):
    def __init__(self):
        super().__init__('conveyor_belt_grasp')
        self.moveit = MoveItPy(node_name='moveit_py')
        self.arm = PlanningComponent(self.moveit, 'arm_group', 'link5')

    def detection_callback(self, msg):
        """收到检测结果，触发抓取"""
        self.get_logger().info(f'检测到物体: ({msg.pose.position.x:.2f}, {msg.pose.position.y:.2f})')

        # 计算抓取位姿
        grasp = deepcopy(msg)
        approach = deepcopy(msg)
        approach.pose.position.z += 0.12

        # 执行抓取
        self.arm.set_start_state_to_current_state()
        self.arm.set_pose_target(approach.pose, 'link5')
        plan = self.arm.plan()
        if plan:
            self.arm.execute(plan.trajectory)

        self.arm.set_start_state_to_current_state()
        self.arm.set_pose_target(grasp.pose, 'link5')
        plan = self.arm.plan()
        if plan:
            self.arm.execute(plan.trajectory)

        # 闭合夹爪
        self.gripper.set_joint_value_target([0.0, 0.0])
        plan = self.gripper.plan()
        if plan:
            self.gripper.execute(plan.trajectory)

        # 撤离
        retreat = deepcopy(grasp)
        retreat.pose.position.z += 0.15
        self.arm.set_pose_target(retreat.pose, 'link5')
        plan = self.arm.plan()
        if plan:
            self.arm.execute(plan.trajectory)

        # 移动到放置位置
        place = PoseStamped()
        place.header.frame_id = 'base_link'
        place.pose.position.x = 0.3
        place.pose.position.y = -0.2
        place.pose.position.z = 0.2

        self.arm.set_pose_target(place.pose, 'link5')
```

### 29.5.2 码垛应用

```python
class PalletizingApp(Node):
    """码垛应用：将物体堆叠到指定位置"""

    def __init__(self):
        super().__init__('palletizing_app')
        self.moveit = MoveItPy(node_name='moveit_py')
        self.arm = PlanningComponent(self.moveit, 'arm_group', 'link5')
        self.gripper = PlanningComponent(self.moveit, 'gripper', 'gripper_link')
        self.psm = self.moveit.get_planning_scene_monitor()

        self.layer = 0
        self.pos_in_layer = 0
        self.layer_positions = [
            [(0.2, -0.1), (0.2, 0.0), (0.2, 0.1)],
            [(0.25, -0.1), (0.25, 0.0), (0.25, 0.1)],
        ]

    def get_next_place_pose(self):
        """获取下一个放置位置（码垛模式）"""
        if self.pos_in_layer >= len(self.layer_positions[self.layer]):
            self.pos_in_layer = 0
            self.layer += 1

        if self.layer >= len(self.layer_positions):
            self.get_logger().warn('码垛已满')
            return None

        x, y = self.layer_positions[self.layer][self.pos_in_layer]
        z = 0.15 + self.layer * 0.1

        pose = PoseStamped()
        pose.header.frame_id = 'base_link'
        pose.pose.position.x = x
        pose.pose.position.y = y
        pose.pose.position.z = z
        pose.pose.orientation.w = 1.0

        self.pos_in_layer += 1
        return pose
```
