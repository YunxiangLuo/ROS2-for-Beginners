## 34.1 视觉抓取管线概述

### 34.1.1 管线架构

视觉抓取是机械臂应用的核心场景，涉及以下关键技术链：

```
相机采集 → 物体检测 → 位姿估计 → 坐标变换 → 运动规划 → 抓取执行
```

**完整流程**：

1. **感知**：相机采集图像，检测目标物体
2. **定位**：获取物体在相机坐标系下的位姿
3. **变换**：通过TF将物体位姿变换到机械臂基座坐标系
4. **规划**：使用MoveIt2进行逆运动学解算和轨迹规划
5. **执行**：控制机械臂完成抓取和放置

**坐标变换链**：

```
object → camera_link → base_link → (MoveIt2) → end_effector
```

关键变换关系：
- camera_link → base_link：手眼标定获得（或URDF固定）
- object → camera_link：物体检测/标记检测获得

## 34.2 视觉检测与定位

### 34.2.1 AR标签检测与定位

基于ArUco标签的视觉抓取是最常用的方案之一。

```python
#!/usr/bin/env python3
"""AR标签抓取检测节点"""
import rclpy
from rclpy.node import Node
from sensor_msgs.msg import Image, CameraInfo
from geometry_msgs.msg import PoseStamped
from cv_bridge import CvBridge
import cv2
import cv2.aruco as aruco
import numpy as np
import tf2_ros
import tf_transformations


class VisionGraspDetector(Node):
    def __init__(self):
        super().__init__('vision_grasp_detector')
        self.bridge = CvBridge()

        # TF2
        self.tf_buffer = tf2_ros.Buffer()
        self.tf_listener = tf2_ros.TransformListener(self.tf_buffer, self)

        # ArUco配置
        dictionary = aruco.getPredefinedDictionary(aruco.DICT_6X6_250)
        params = aruco.DetectorParameters()
        self.detector = aruco.ArucoDetector(dictionary, params)

        self.declare_parameter('marker_size', 0.065)
        self.marker_size = self.get_parameter('marker_size').value

        self.camera_matrix = None
        self.dist_coeffs = None

        # 发布者：目标位姿（base_link坐标系）
        self.pose_pub = self.create_publisher(
            PoseStamped, '/target_pose', 10
        )

        # 订阅者
        self.image_sub = self.create_subscription(
            Image, '/camera/color/image_raw',
            self.image_callback, 10
        )
        self.info_sub = self.create_subscription(
            CameraInfo, '/camera/color/camera_info',
            self.info_callback, 10
        )

    def info_callback(self, msg):
        if self.camera_matrix is None:
            self.camera_matrix = np.array(msg.k).reshape(3, 3)
            self.dist_coeffs = np.array(msg.d)

    def image_callback(self, msg):
        if self.camera_matrix is None:
            return

        frame = self.bridge.imgmsg_to_cv2(msg, 'bgr8')
        gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)

        corners, ids, _ = self.detector.detectMarkers(gray)

        if ids is not None:
            rvecs, tvecs, _ = aruco.estimatePoseSingleMarkers(
                corners, self.marker_size,
                self.camera_matrix, self.dist_coeffs
            )

            for i in range(len(ids)):
                # 物体在相机坐标系下的位姿
                pose_cam = PoseStamped()
                pose_cam.header.stamp = self.get_clock().now().to_msg()
                pose_cam.header.frame_id = 'camera_color_optical_frame'
                pose_cam.pose.position.x = float(tvecs[i][0][0])
                pose_cam.pose.position.y = float(tvecs[i][0][1])
                pose_cam.pose.position.z = float(tvecs[i][0][2])

                rmat, _ = cv2.Rodrigues(rvecs[i])
                q = tf_transformations.quaternion_from_matrix(
                    np.vstack([
                        np.hstack([rmat, [[0], [0], [0]]]),
                        [0, 0, 0, 1]
                    ])
                )
                pose_cam.pose.orientation.x = q[0]
                pose_cam.pose.orientation.y = q[1]
                pose_cam.pose.orientation.z = q[2]
                pose_cam.pose.orientation.w = q[3]

                try:
                    # 变换到base_link坐标系
                    transform = self.tf_buffer.lookup_transform(
                        'base_link',
                        'camera_color_optical_frame',
                        rclpy.time.Time()
                    )

                    # 使用tf2_geometry_msgs进行坐标变换
                    from tf2_geometry_msgs import do_transform_pose
                    pose_base = do_transform_pose(
                        pose_cam.pose, transform
                    )

                    result = PoseStamped()
                    result.header.stamp = self.get_clock().now().to_msg()
                    result.header.frame_id = 'base_link'
                    result.pose = pose_base

                    self.pose_pub.publish(result)

                    self.get_logger().info(
                        f'目标 {ids[i][0]}: '
                        f'({result.pose.position.x:.3f}, '
                        f'{result.pose.position.y:.3f}, '
                        f'{result.pose.position.z:.3f})'
                    )

                except Exception as e:
                    self.get_logger().warn(f'TF变换失败: {e}')

        cv2.imshow('Vision Grasp', frame)
        cv2.waitKey(1)


def main(args=None):
    rclpy.init(args=args)
    node = VisionGraspDetector()
    try:
        rclpy.spin(node)
    except KeyboardInterrupt:
        cv2.destroyAllWindows()
    finally:
        node.destroy_node()
        rclpy.shutdown()


if __name__ == '__main__':
    main()
```

### 34.2.2 TF坐标变换

```python
class TFTransformer:
    """TF坐标变换工具"""

    def __init__(self, node):
        self.tf_buffer = tf2_ros.Buffer()
        self.tf_listener = tf2_ros.TransformListener(self.tf_buffer, node)

    def transform_pose(self, pose_stamped, target_frame):
        """将位姿变换到目标坐标系"""
        from tf2_geometry_msgs import do_transform_pose

        try:
            transform = self.tf_buffer.lookup_transform(
                target_frame,
                pose_stamped.header.frame_id,
                rclpy.time.Time(),
                rclpy.duration.Duration(seconds=1.0)
            )

            transformed = do_transform_pose(pose_stamped.pose, transform)

            result = PoseStamped()
            result.header.stamp = pose_stamped.header.stamp
            result.header.frame_id = target_frame
            result.pose = transformed
            return result

        except Exception as e:
            print(f'TF变换失败: {e}')
            return None

    def lookup_transform(self, target_frame, source_frame):
        """查询坐标变换"""
        try:
            return self.tf_buffer.lookup_transform(
                target_frame, source_frame,
                rclpy.time.Time(),
                rclpy.duration.Duration(seconds=1.0)
            )
        except Exception as e:
            print(f'TF查询失败: {e}')
            return None

    def transform_point(self, point_stamped, target_frame):
        """变换点坐标"""
        from tf2_geometry_msgs import do_transform_point

        try:
            transform = self.tf_buffer.lookup_transform(
                target_frame,
                point_stamped.header.frame_id,
                rclpy.time.Time()
            )
            transformed = do_transform_point(point_stamped.point, transform)
            return transformed
        except Exception as e:
            print(f'点变换失败: {e}')
            return None
```

## 34.3 MoveIt2 + 视觉融合

### 34.3.1 抓取规划节点

```python
#!/usr/bin/env python3
"""MoveIt2抓取规划节点 - 接收视觉检测结果"""
import rclpy
from rclpy.node import Node
from geometry_msgs.msg import PoseStamped
from moveit.planning import MoveItPy, PlanningComponent
import numpy as np


class MoveItGraspPlanner(Node):
    def __init__(self):
        super().__init__('moveit_grasp_planner')
        self.moveit = MoveItPy(node_name='moveit_py')
        self.arm = PlanningComponent(self.moveit, 'arm_group', 'link5')
        self.gripper = PlanningComponent(
            self.moveit, 'gripper', 'gripper_link'
        )

        # 配置规划参数
        self.arm.set_goal_position_tolerance(0.01)
        self.arm.set_goal_orientation_tolerance(0.02)
        self.arm.set_max_velocity_scaling_factor(0.4)
        self.arm.set_max_acceleration_scaling_factor(0.3)

        # 订阅检测结果
        self.target_pose_sub = self.create_subscription(
            PoseStamped, '/target_pose',
            self.target_callback, 10
        )

        self.get_logger().info('抓取规划节点已启动')

    def target_callback(self, msg):
        """收到目标位姿，触发抓取"""
        self.get_logger().info(
            f'收到抓取目标: ({msg.pose.position.x:.3f}, '
            f'{msg.pose.position.y:.3f}, {msg.pose.position.z:.3f})'
        )
        self.execute_grasp(msg)

    def execute_grasp(self, target_pose_msg):
        """执行抓取"""
        # 1. 打开夹爪
        self.open_gripper()

        # 2. 移动到预抓取位姿（目标上方10cm）
        pre_grasp = PoseStamped()
        pre_grasp.header = target_pose_msg.header
        pre_grasp.pose = target_pose_msg.pose
        pre_grasp.pose.position.z += 0.12

        self.arm.set_start_state_to_current_state()
        self.arm.set_pose_target(pre_grasp.pose, 'link5')
        plan = self.arm.plan(
            planner_id='RRTConnectkConfigDefault',
            planning_time=2.0
        )
        if plan:
            self.get_logger().info('预抓取规划成功')
            self.arm.execute(plan.trajectory)
        else:
            self.get_logger().error('预抓取规划失败')
            return

        # 3. 下降到抓取位姿
        self.arm.set_start_state_to_current_state()
        self.arm.set_pose_target(target_pose_msg.pose, 'link5')
        plan = self.arm.plan(
            planner_id='RRTConnectkConfigDefault',
            planning_time=2.0
        )
        if plan:
            self.get_logger().info('抓取规划成功')
            self.arm.execute(plan.trajectory)
        else:
            self.get_logger().error('抓取规划失败')
            return

        # 4. 闭合夹爪
        self.close_gripper()

        # 5. 撤离
        retreat = PoseStamped()
        retreat.header = target_pose_msg.header
        retreat.pose = target_pose_msg.pose
        retreat.pose.position.z += 0.15

        self.arm.set_start_state_to_current_state()
        self.arm.set_pose_target(retreat.pose, 'link5')
        plan = self.arm.plan()
        if plan:
            self.arm.execute(plan.trajectory)

        self.get_logger().info('抓取完成')

    def open_gripper(self):
        """打开夹爪"""
        self.gripper.set_start_state_to_current_state()
        self.gripper.set_joint_value_target([0.65, 0.65])
        plan = self.gripper.plan()
        if plan:
            self.gripper.execute(plan.trajectory)

    def close_gripper(self):
        """闭合夹爪"""
        self.gripper.set_start_state_to_current_state()
        self.gripper.set_joint_value_target([0.0, 0.0])
        plan = self.gripper.plan()
        if plan:
            self.gripper.execute(plan.trajectory)


def main(args=None):
    rclpy.init(args=args)
    node = MoveItGraspPlanner()
    try:
        rclpy.spin(node)
    except KeyboardInterrupt:
        pass
    finally:
        node.destroy_node()
        rclpy.shutdown()


if __name__ == '__main__':
    main()
```

## 34.4 完整Pick-and-Place流程

### 34.4.1 综合抓取放置

```python
#!/usr/bin/env python3
"""完整视觉抓取放置流程"""
import time
import rclpy
from rclpy.node import Node
from geometry_msgs.msg import PoseStamped
from moveit.planning import MoveItPy, PlanningComponent
from moveit_msgs.msg import (
    CollisionObject, AttachedCollisionObject
)
from shape_msgs.msg import SolidPrimitive


class VisualPickAndPlace(Node):
    """视觉引导的完整Pick-and-Place"""

    def __init__(self):
        super().__init__('visual_pick_and_place')
        self.moveit = MoveItPy(node_name='moveit_py')
        self.psm = self.moveit.get_planning_scene_monitor()
        self.arm = PlanningComponent(self.moveit, 'arm_group', 'link5')
        self.gripper = PlanningComponent(
            self.moveit, 'gripper', 'gripper_link'
        )

        self.arm.set_goal_position_tolerance(0.01)
        self.arm.set_goal_orientation_tolerance(0.02)

        # 设置场景
        self.setup_scene()

        # 订阅检测结果
        self.sub = self.create_subscription(
            PoseStamped, '/target_pose',
            self.pick_and_place_callback, 10
        )

        self.get_logger().info('视觉抓取放置节点已启动')

    def setup_scene(self):
        """初始化规划场景"""
        # 添加桌子
        table = CollisionObject()
        table.id = 'table'
        table.header.frame_id = 'base_link'
        table.operation = CollisionObject.ADD
        prim = SolidPrimitive()
        prim.type = SolidPrimitive.BOX
        prim.dimensions = [0.8, 0.6, 0.02]
        table.primitives = [prim]
        pose = PoseStamped()
        pose.header.frame_id = 'base_link'
        pose.pose.position.z = -0.01
        pose.pose.orientation.w = 1.0
        table.primitive_poses = [pose.pose]
        self.psm.process_collision_object(table)
        time.sleep(0.5)

    def pick_and_place_callback(self, msg):
        """收到目标位姿执行抓取放置"""
        self.get_logger().info('开始抓取放置流程')

        # 步骤1-5：抓取
        if not self.pick(msg):
            self.get_logger().error('抓取失败')
            return

        time.sleep(1)

        # 步骤6-10：放置
        place_pose = PoseStamped()
        place_pose.header.frame_id = 'base_link'
        place_pose.pose.position.x = 0.3
        place_pose.pose.position.y = -0.2
        place_pose.pose.position.z = 0.15
        place_pose.pose.orientation.w = 1.0

        if not self.place(place_pose):
            self.get_logger().error('放置失败')
            return

        self.get_logger().info('抓取放置流程完成')

    def pick(self, target):
        """抓取物体"""
        # 步骤1：打开夹爪
        self.get_logger().info('打开夹爪')
        self.gripper.set_start_state_to_current_state()
        self.gripper.set_joint_value_target([0.65, 0.65])
        plan = self.gripper.plan()
        if plan:
            self.gripper.execute(plan.trajectory)

        # 步骤2：移动到预抓取位姿
        pre_grasp = PoseStamped()
        pre_grasp.header = target.header
        pre_grasp.pose = target.pose
        pre_grasp.pose.position.z += 0.12

        self.get_logger().info('移动到预抓取位姿')
        self.arm.set_start_state_to_current_state()
        self.arm.set_pose_target(pre_grasp.pose, 'link5')
        plan = self.arm.plan()
        if not plan:
            return False
        self.arm.execute(plan.trajectory)

        # 步骤3：下降到抓取位姿
        self.get_logger().info('下降到抓取位姿')
        self.arm.set_start_state_to_current_state()
        self.arm.set_pose_target(target.pose, 'link5')
        plan = self.arm.plan()
        if not plan:
            return False
        self.arm.execute(plan.trajectory)

        # 步骤4：闭合夹爪
        self.get_logger().info('闭合夹爪')
        self.gripper.set_start_state_to_current_state()
        self.gripper.set_joint_value_target([0.0, 0.0])
        plan = self.gripper.plan()
        if plan:
            self.gripper.execute(plan.trajectory)

        # 步骤5：附着物体
        self.get_logger().info('附着物体')
        self.attach_object('target_object')

        # 步骤6：撤离
        retreat = PoseStamped()
        retreat.header = target.header
        retreat.pose = target.pose
        retreat.pose.position.z += 0.15

        self.get_logger().info('撤离')
        self.arm.set_start_state_to_current_state()
        self.arm.set_pose_target(retreat.pose, 'link5')
        plan = self.arm.plan()
        if not plan:
            return False
        self.arm.execute(plan.trajectory)

        return True

    def place(self, place_pose):
        """放置物体"""
        # 步骤7：移动到预放置位姿
        pre_place = PoseStamped()
        pre_place.header = place_pose.header
        pre_place.pose = place_pose.pose
        pre_place.pose.position.z += 0.12

        self.get_logger().info('移动到预放置位姿')
        self.arm.set_start_state_to_current_state()
        self.arm.set_pose_target(pre_place.pose, 'link5')
        plan = self.arm.plan()
        if not plan:
            return False
        self.arm.execute(plan.trajectory)

        # 步骤8：下降到放置位姿
        self.get_logger().info('下降到放置位姿')
        self.arm.set_start_state_to_current_state()
        self.arm.set_pose_target(place_pose.pose, 'link5')
        plan = self.arm.plan()
        if not plan:
            return False
        self.arm.execute(plan.trajectory)

        # 步骤9：打开夹爪
        self.get_logger().info('打开夹爪释放物体')
        self.gripper.set_start_state_to_current_state()
        self.gripper.set_joint_value_target([0.65, 0.65])
        plan = self.gripper.plan()
        if plan:
            self.gripper.execute(plan.trajectory)

        # 步骤10：分离物体
        self.get_logger().info('分离物体')
        self.detach_object('target_object')

        # 步骤11：撤离
        retreat = PoseStamped()
        retreat.header = place_pose.header
        retreat.pose = place_pose.pose
        retreat.pose.position.z += 0.15

        self.get_logger().info('撤离')
        self.arm.set_start_state_to_current_state()
        self.arm.set_pose_target(retreat.pose, 'link5')
        plan = self.arm.plan()
        if not plan:
            return False
        self.arm.execute(plan.trajectory)

        return True

    def attach_object(self, object_id):
        """附着物体到末端"""
        co = CollisionObject()
        co.id = object_id
        co.header.frame_id = 'gripper_link'
        co.operation = CollisionObject.ADD

        primitive = SolidPrimitive()
        primitive.type = SolidPrimitive.BOX
        primitive.dimensions = [0.04, 0.04, 0.08]

        pose = PoseStamped()
        pose.header.frame_id = 'gripper_link'
        pose.pose.position.z = 0.04
        pose.pose.orientation.w = 1.0

        co.primitives = [primitive]
        co.primitive_poses = [pose.pose]

        aco = AttachedCollisionObject()
        aco.link_name = 'gripper_link'
        aco.object = co
        aco.touch_links = ['gripper_link', 'link5']

        self.psm.process_attached_collision_object(aco)

    def detach_object(self, object_id):
        """分离物体"""
        co = CollisionObject()
        co.id = object_id
        co.operation = CollisionObject.REMOVE

        aco = AttachedCollisionObject()
        aco.link_name = 'gripper_link'
        aco.object = co

        self.psm.process_attached_collision_object(aco)


def main(args=None):
```

```python
        # 抓取

        place_pose.pose.position.y = zone[1]
        place_pose.pose.position.z = zone[2]
        place_pose.pose.orientation.w = 1.0

        # 放置
        self.place(place_pose)

    def run_batch(self, objects):
        """批量分拣"""
        for obj in objects:
            self.sort_object(obj['pose'], obj['color'])
            time.sleep(0.5)
```

```python
class YOLOGraspBridge(Node):
    """YOLO检测结果 → 抓取位姿"""

    def __init__(self):
        super().__init__('yolo_grasp_bridge')
        self.bridge = CvBridge()

        # 相机内参（需标定）
        self.fx = 600.0
        self.fy = 600.0
        self.cx = 320.0
        self.cy = 240.0
        self.depth = 0.5  # 假设深度

        self.det_sub = self.create_subscription(
            Detection2DArray, '/yolo/detections',
            self.detection_callback, 10
        )
        self.pose_pub = self.create_publisher(
            PoseStamped, '/target_pose', 10
        )

    def pixel_to_3d(self, u, v, depth):
        """像素坐标 → 3D坐标"""
        x = (u - self.cx) * depth / self.fx
        y = (v - self.cy) * depth / self.fy
        z = depth
        return x, y, z

    def detection_callback(self, msg):
        for det in msg.detections:
            # 检测框中心
            cx = det.bbox.center.x
            cy = det.bbox.center.y

            # 获取深度（默认值或深度图）
            depth = self.get_depth_at(cx, cy)

            # 像素 → 3D
            x, y, z = self.pixel_to_3d(cx, cy, depth)

            # 发布抓取位姿
            pose = PoseStamped()
            pose.header = det.header
            pose.pose.position.x = x
            pose.pose.position.y = y
            pose.pose.position.z = z
            pose.pose.orientation.w = 1.0

            self.pose_pub.publish(pose)

    def get_depth_at(self, u, v):
        """获取指定像素的深度（简化版）"""
        # 实际应订阅深度图话题
        return self.depth
```

### 34.6.1 完整AR抓取系统

```python
#!/usr/bin/env python3
"""AR标签引导的完整抓取系统"""
import rclpy
from rclpy.node import Node
from geometry_msgs.msg import PoseStamped
import yaml
import tf2_ros
import tf_transformations
import numpy as np


class ARGraspSystem(Node):
    """AR标签抓取系统 - 整合检测+变换+规划"""

    def __init__(self):
        super().__init__('ar_grasp_system')
        self.tf_buffer = tf2_ros.Buffer()
        self.tf_listener = tf2_ros.TransformListener(self.tf_buffer, self)

        # 加载标定结果
        self.load_calibration('handeye_calibration.yaml')

        # 检测到的目标缓存
        self.current_target = None

        # 订阅AR标签位姿
        self.pose_sub = self.create_subscription(
            PoseStamped, '/aruco/pose',
            self.aruco_callback, 10
        )

        self.get_logger().info('AR抓取系统已启动')

    def load_calibration(self, calib_file):
        """加载手眼标定结果"""
        try:
            with open(calib_file, 'r') as f:
                calib = yaml.safe_load(f)
            self.get_logger().info('手眼标定结果已加载')
        except FileNotFoundError:
            self.get_logger().warn('未找到标定文件，使用默认值')

    def aruco_callback(self, msg):
        """收到ArUco标签位姿"""
        try:
            # 变换到base_link
            transform = self.tf_buffer.lookup_transform(
                'base_link',
                msg.header.frame_id,
                rclpy.time.Time()
            )

            from tf2_geometry_msgs import do_transform_pose
            pose_base = do_transform_pose(msg.pose, transform)

            # 添加抓取偏移
            pose_base.position.z += 0.03  # 物体高度补偿

            self.current_target = {
                'pose': pose_base,
                'stamp': self.get_clock().now(),
            }

            self.get_logger().info(
                f'目标在base_link下: '
                f'({pose_base.position.x:.3f}, '
                f'{pose_base.position.y:.3f}, '
                f'{pose_base.position.z:.3f})'
            )

        except Exception as e:
            self.get_logger().error(f'位姿变换失败: {e}')

    def get_target(self):
        """获取当前目标"""
        if self.current_target is None:
            return None

        # 检查目标是否过期（超过2秒）
        elapsed = (self.get_clock().now() -
                   self.current_target['stamp']).nanoseconds / 1e9
        if elapsed > 2.0:
            self.current_target = None
            return None

        return self.current_target['pose']


def main(args=None):
    rclpy.init(args=args)
    node = ARGraspSystem()
    rclpy.spin(node)
    node.destroy_node()
    rclpy.shutdown()


if __name__ == '__main__':
    main()
```
