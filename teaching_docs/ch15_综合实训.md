## 15.2 题目1：LLM 配方验证服务

### 知识点
- 服务通信（Service）：Client 发送配方文本，Server 返回验证结果
- LLM Prompt Engineering：设计化学配比校验 prompt
- 结构化输出解析

### 核心代码

```python
#!/usr/bin/env python3
"""recipe_validator: LLM 校验化学实验配方 — Service 节点"""
import json
import rclpy
from rclpy.node import Node
from example_interfaces.srv import Trigger  # 简化：输入 string, 输出 result

# 实际使用自定义 .srv: RecipeValidate.srv
# string recipe_text
# ---
# bool is_valid
# string feedback
# string corrected_formula

class RecipeValidator(Node):
    def __init__(self):
        super().__init__('recipe_validator')
        self.srv = self.create_service(
            Trigger, 'validate_recipe', self.validate_callback)

    def validate_callback(self, request, response):
        recipe = request.data  # 学生输入的配方文本
        # 调用 LLM 校验
        prompt = self.build_validation_prompt(recipe)
        llm_result = self.call_llm(prompt)
        parsed = self.parse_llm_response(llm_result)

        response.success = parsed.get('is_valid', False)
        response.message = json.dumps(parsed, ensure_ascii=False)
        return response

    def build_validation_prompt(self, recipe_text):
        """构建化学配比校验 prompt"""
        return f"""你是一位中学化学实验安全专家。请校验以下实验配方的正确性：

{recipe_text}

请验证：
1. 各试剂配比是否符合化学计量关系
2. 预期产物是否正确
3. 是否存在安全隐患

返回 JSON 格式：
{{"is_valid": bool, "feedback": "校验意见", "products": ["产物1"], "safety_warnings": ["警告"]}}
"""

    def call_llm(self, prompt):
        # 调用 OpenAI / Qwen API（简化示例）
        try:
            from openai import OpenAI
            client = OpenAI()
            resp = client.chat.completions.create(
                model="gpt-4o-mini",
                messages=[{"role": "user", "content": prompt}],
                temperature=0.1
            )
            return resp.choices[0].message.content
        except ImportError:
            return '{"is_valid": true, "feedback": "LLM未配置，默认通过"}'

    def parse_llm_response(self, text):
        try:
            return json.loads(text)
        except json.JSONDecodeError:
            return {"is_valid": True, "feedback": text[:200]}


def main(args=None):
    rclpy.init(args=args)
    rclpy.spin(RecipeValidator())
    rclpy.shutdown()
```

```python
#!/usr/bin/env python3
"""bottle_detector: YOLO 检测 Gazebo 中的试剂瓶 — 话题节点"""
import cv2
import rclpy
from rclpy.node import Node
from sensor_msgs.msg import Image
from vision_msgs.msg import Detection2DArray
from cv_bridge import CvBridge

# 自定义检测话题: /bottle_detections (Detection2DArray)


class BottleDetector(Node):
    def __init__(self):
        super().__init__('bottle_detector')
        self.bridge = CvBridge()
        # 订阅 XBot-U 相机图像
        self.sub = self.create_subscription(
            Image, '/camera/image_raw', self.image_callback, 10)
        # 发布检测结果
        self.det_pub = self.create_publisher(
            Detection2DArray, '/bottle_detections', 10)
        # 加载 YOLO 模型
        self.model = self.load_model()

    def load_model(self):
        try:
            from ultralytics import YOLO
            return YOLO('yolov8n.pt')  # 或自定义训练的试剂瓶模型
        except ImportError:
            self.get_logger().warn('ultralytics 未安装，使用模拟检测')
            return None

    def image_callback(self, msg):
        # ROS Image → OpenCV
        cv_image = self.bridge.imgmsg_to_cv2(msg, 'bgr8')

        if self.model is not None:
            results = self.model(cv_image, conf=0.5)
            detections = self.results_to_ros(results)
        else:
            detections = self.simulate_detection(cv_image)

        self.det_pub.publish(detections)

    def results_to_ros(self, results):
        """YOLO 结果 → ROS Detection2DArray"""
        det_array = Detection2DArray()
        det_array.header.frame_id = 'camera_link'
        det_array.header.stamp = self.get_clock().now().to_msg()

        for r in results:
            for box in r.boxes:
                cls_name = r.names[int(box.cls[0])]
                self.get_logger().info(f'检测到: {cls_name}')
                # 添加 detection 到数组（简化）
        return det_array

    def simulate_detection(self, image):
        """模拟检测（YOLO 未安装时的降级模式）"""
        det_array = Detection2DArray()
        det_array.header.frame_id = 'camera_link'

    rclpy.init(args=args)
```

```python
#!/usr/bin/env python3
"""label_reader: VLM 读取试剂瓶标签文字 — 服务 + 话题节点"""
import json
import base64
import cv2
import rclpy
from rclpy.node import Node
from sensor_msgs.msg import Image
from cv_bridge import CvBridge


class LabelReader(Node):
    def __init__(self):
        super().__init__('label_reader')
        self.bridge = CvBridge()
        self.sub = self.create_subscription(
            Image, '/camera/image_raw', self.image_cb, 10)

    def image_cb(self, msg):
        # 订阅原始图像并缓存最新一帧
        self.latest_image = self.bridge.imgmsg_to_cv2(msg, 'bgr8')

    def read_label_from_roi(self, roi_image, expected_material):
        """VLM 读取瓶身标签，比对期望材料名称"""
        # 将 roi 图像编码为 base64
        _, buffer = cv2.imencode('.jpg', roi_image)
        img_b64 = base64.b64encode(buffer).decode('utf-8')

        prompt = f"""请读取图中试剂瓶标签上的文字。
期望材料名称: {expected_material}
请返回 JSON: {{"label_text": "读取到的文字", "matches_expected": bool, "confidence": float}}
"""

        import openai
        client = openai.OpenAI()
        resp = client.chat.completions.create(
            model="gpt-4o",
            messages=[{
                "role": "user",
                "content": [
                    {"type": "text", "text": prompt},
                    {"type": "image_url", "image_url": {
                        "url": f"data:image/jpeg;base64,{img_b64}"}}
                ]
            }],
            temperature=0.1
        )
        result = resp.choices[0].message.content
        return json.loads(result)


def main(args=None):
    rclpy.init(args=args)
    rclpy.spin(LabelReader())
```

## 15.5 题目4：TF 空间定位

### 知识点
- TF2 坐标变换：camera_link → marker → base_link
- AR 标签到 TF 变换的转换
- 话题通信：发布检测到的标签位姿

### 核心代码

```python
#!/usr/bin/env python3
"""bottle_localizer: TF 解算试剂瓶 3D 位姿"""
import rclpy
from rclpy.node import Node
from geometry_msgs.msg import PoseStamped, TransformStamped
from tf2_ros import TransformBroadcaster, Buffer, TransformListener


class BottleLocalizer(Node):
    def __init__(self):
        super().__init__('bottle_localizer')
        self.tf_buffer = Buffer()
        self.tf_listener = TransformListener(self.tf_buffer, self)
        self.tf_broadcaster = TransformBroadcaster(self)
        self.timer = self.create_timer(0.5, self.localize_bottles)

    def localize_bottles(self):
        """遍历已知的 AR 标签 frame，计算全局位姿"""
        marker_frames = ['ar_marker_hcl', 'ar_marker_naoh',
                         'ar_marker_h2o', 'ar_marker_phenolphthalein']

        for frame_id in marker_frames:
            try:
                # 查询标签相对于 base_link 的位姿
                t = self.tf_buffer.lookup_transform(
                    'base_link', frame_id, rclpy.time.Time(),
                    rclpy.duration.Duration(seconds=1.0))

                # 发布试剂瓶的 3D 位姿（用于 MoveIt2 抓取）
                pose = PoseStamped()
                pose.header.frame_id = 'base_link'
                pose.header.stamp = self.get_clock().now().to_msg()
                pose.pose.position.x = t.transform.translation.x
                pose.pose.position.y = t.transform.translation.y
                pose.pose.position.z = t.transform.translation.z + 0.15
                pose.pose.orientation = t.transform.rotation

                self.get_logger().info(
                    f'{frame_id} 位姿: '
                    f'({pose.pose.position.x:.3f}, '
                    f'{pose.pose.position.y:.3f}, '
                    f'{pose.pose.position.z:.3f})')

            except Exception as e:
                self.get_logger().debug(
                    f'{frame_id} 未找到: {e}')

    def broadcast_marker_transform(self, marker_id, x, y, z):
        """广播 AR 标签的静态 TF（仿真环境中模拟）"""
        t = TransformStamped()
        t.header.stamp = self.get_clock().now().to_msg()
        t.header.frame_id = 'base_link'
        t.child_frame_id = f'ar_marker_{marker_id}'
        t.transform.translation.x = x
        t.transform.translation.y = y
        t.transform.translation.z = z
        t.transform.rotation.w = 1.0
        self.tf_broadcaster.sendTransform(t)


def main(args=None):
    rclpy.init(args=args)
    rclpy.spin(BottleLocalizer())
    rclpy.shutdown()
```

## 15.7 题目6：全流程编排（Action Server）

### 知识点
- Action Server 编排多步骤任务
- 服务调用链：验证 → 检测 → 识别 → 定位 → 抓取
- 循环迭代所有配方组分
- 错误处理和状态恢复

### 核心代码

```python
#!/usr/bin/env python3
"""experiment_pipeline: 化学实验全流程编排 — Action Server"""
import asyncio
import json
import rclpy
from rclpy.node import Node
from rclpy.action import ActionServer, GoalResponse, CancelResponse


# 自定义 Action: ExperimentPipeline.action
# string recipe_text         # 配方文本
# ---
# bool success               # 全部完成
# string error_message       # 错误信息
# ---
# int32 current_step         # 当前步骤
# int32 total_steps          # 总步骤数
# string status_text         # 状态描述


class ExperimentPipeline(Node):
    def __init__(self):
        super().__init__('experiment_pipeline')
        self.action_server = ActionServer(
            self, ExperimentPipeline, 'run_experiment',
            execute_callback=self.execute,
            goal_callback=self.goal_cb,
            cancel_callback=self.cancel_cb)

        # 创建各模块的客户端
        self.validator_client = self.create_client(
            Trigger, 'validate_recipe')
        # ... 其他客户端

    def goal_cb(self, goal):
        self.get_logger().info(
            f'收到实验配方: {goal.recipe_text[:50]}...')
        return GoalResponse.ACCEPT

    def cancel_cb(self, goal_handle):
        self.get_logger().warn('实验被取消！')
        return CancelResponse.ACCEPT

    async def execute(self, goal_handle):
        recipe = goal_handle.request.recipe_text
        fb = ExperimentPipeline.Feedback()
        total = 6
        fb.total_steps = total

        try:
            # 步骤1：LLM 校验配方
            fb.current_step = 1
            fb.status_text = 'Step 1/6: 校验配方配比...'
            goal_handle.publish_feedback(fb)
            valid = await self.validate_recipe(recipe)
            if not valid:
                goal_handle.abort()
                return ExperimentPipeline.Result(
                    success=False,
                    error_message='配方校验失败')

            # 步骤2：获取配方组分列表
            fb.current_step = 2
            fb.status_text = 'Step 2/6: 解析配方组分列表...'
            goal_handle.publish_feedback(fb)
            components = self.parse_components(recipe)

            # 步骤3-5：循环处理每个组分
            for i, comp in enumerate(components):
                if goal_handle.is_cancel_requested:
                    goal_handle.canceled()
                    return ExperimentPipeline.Result()

                # 3. 检测试剂瓶
                fb.current_step = 3
                fb.status_text = (
                    f'Step 3/6: YOLO 检测 {comp["name"]} 瓶... '
                    f'({i+1}/{len(components)})')
                goal_handle.publish_feedback(fb)
                detection = await self.detect_bottle(comp['name'])

                # 4. VLM 验证标签
                fb.current_step = 4
                fb.status_text = (
                    f'Step 4/6: VLM 识别 {comp["name"]} 标签...')
                goal_handle.publish_feedback(fb)
                verified = await self.verify_label(
                    detection.image, comp['name'])
                if not verified:
                    continue

                # 5. 定位并抓取
                fb.current_step = 5
                fb.status_text = (
                    f'Step 5/6: 定位+抓取 {comp["name"]}...')
                goal_handle.publish_feedback(fb)
                pose = await self.localize_bottle(comp['name'])
                await self.pick_and_place(pose, comp['name'])

                await asyncio.sleep(0.5)

            # 步骤6：完成
            fb.current_step = 6
            fb.status_text = (
                'Step 6/6: ✅ 实验配制完成！')
            goal_handle.publish_feedback(fb)
            goal_handle.succeed()
            return ExperimentPipeline.Result(success=True)

        except Exception as e:
            goal_handle.abort()
            return ExperimentPipeline.Result(
                success=False, error_message=str(e))

    async def validate_recipe(self, recipe):
        """调用 LLM 验证服务"""
        req = Trigger.Request()
        req.data = recipe
        future = self.validator_client.call_async(req)
        await future
        result = json.loads(future.result().message)
        self.get_logger().info(
            f'校验结果: valid={result.get("is_valid")}')
        return result.get('is_valid', False)

    def parse_components(self, recipe):
        """解析配方为组分列表"""
        return [
            {"name": "HCl", "volume_ml": 5},
            {"name": "NaOH", "volume_ml": 5},
            {"name": "Phenolphthalein", "volume_ml": 2},
```
