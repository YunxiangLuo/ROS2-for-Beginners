## 35.2 系统架构设计

### 35.2.1 架构设计原则

架构设计遵循四条原则：一是模块化，每个功能独立为一个节点，通过ROS2话题/服务/动作通信；二是可扩展，支持添加新的传感器或执行器；三是容错性，关键节点有错误处理和降级策略；四是实时性，规划和控制使用动作通信。

### 35.3.1 模块1：YOLO目标检测模块

```python
#!/usr/bin/env python3
"""模块1：YOLO目标检测节点"""
import rclpy
from rclpy.node import Node
from sensor_msgs.msg import Image
from vision_msgs.msg import Detection2D, Detection2DArray
from std_msgs.msg import Header
from cv_bridge import CvBridge
import cv2

class ProjectYOLODetector(Node):
    """综合实训 - YOLO检测模块"""
    def __init__(self):
        super().__init__('project_yolo_detector')
        self.bridge = CvBridge()

        self.declare_parameter('model_path', 'yolov8n.pt')
        self.declare_parameter('confidence', 0.5)
        model_path = self.get_parameter('model_path').value
        self.confidence = self.get_parameter('confidence').value

        # 加载YOLO模型
        try:
            from ultralytics import YOLO
            self.model = YOLO(model_path)
            self.get_logger().info(f'YOLO模型加载成功: {model_path}')
        except Exception as e:
            self.get_logger().error(f'模型加载失败: {e}')
            self.model = None

        # 订阅相机图像
        self.image_sub = self.create_subscription(
            Image, '/camera/image_raw', self.image_callback, 10
        )
        # 发布检测结果
        self.det_pub = self.create_publisher(
            Detection2DArray, '/yolo/detections', 10
        )
        # 发布可视化结果
        self.vis_pub = self.create_publisher(
            Image, '/yolo/visualization', 10
        )
        self.get_logger().info('YOLO检测模块已启动')

    def image_callback(self, msg):
        if self.model is None:
            return

        frame = self.bridge.imgmsg_to_cv2(msg, 'bgr8')
        results = self.model(frame, conf=self.confidence, verbose=False)[0]

        detections_msg = Detection2DArray()
        detections_msg.header = Header(
            stamp=self.get_clock().now().to_msg(),
            frame_id=msg.header.frame_id
        )

        annotated = frame.copy()
        if results.boxes is not None:
            for box in results.boxes:
                x1, y1, x2, y2 = box.xyxy[0].tolist()
                conf = float(box.conf[0])
                cls_id = int(box.cls[0])
                cls_name = results.names[cls_id]

                detection = Detection2D()
                detection.header = detections_msg.header
                detection.bbox.center.x = float((x1 + x2) / 2)
                detection.bbox.center.y = float((y1 + y2) / 2)
                detection.bbox.size_x = float(x2 - x1)
                detection.bbox.size_y = float(y2 - y1)
                detection.results[0].hypothesis.class_id = cls_name
                detection.results[0].hypothesis.score = conf
                detections_msg.detections.append(detection)

                cv2.rectangle(annotated, (int(x1), int(y1)),
                              (int(x2), int(y2)), (0, 255, 0), 2)
                cv2.putText(annotated, f'{cls_name} {conf:.2f}',
                            (int(x1), int(y1)-5),
                            cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 255, 0), 2)

        self.det_pub.publish(detections_msg)
        vis_msg = self.bridge.cv2_to_imgmsg(annotated, 'bgr8')
        vis_msg.header = msg.header
        self.vis_pub.publish(vis_msg)


def main(args=None):
    rclpy.init(args=args)
    node = ProjectYOLODetector()
    rclpy.spin(node)
    node.destroy_node()
    rclpy.shutdown()


if __name__ == '__main__':
    main()
```

### 35.3.2 模块2：AR标签定位模块

```python
#!/usr/bin/env python3
"""模块2：AR标签定位 - 获取目标3D位姿"""
import rclpy
from rclpy.node import Node
from sensor_msgs.msg import Image, CameraInfo
from geometry_msgs.msg import PoseStamped
from cv_bridge import CvBridge
import cv2
import cv2.aruco as aruco
import numpy as np
import tf2_ros
from tf2_geometry_msgs import do_transform_pose

class ProjectARLocalizer(Node):
    """综合实训 - AR标签定位模块"""
    def __init__(self):
        super().__init__('project_ar_localizer')
        self.bridge = CvBridge()
        self.tf_buffer = tf2_ros.Buffer()
        self.tf_listener = tf2_ros.TransformListener(self.tf_buffer, self)

        dictionary = aruco.getPredefinedDictionary(aruco.DICT_6X6_250)
        self.detector = aruco.ArucoDetector(dictionary)

        self.declare_parameter('marker_size', 0.065)
        self.marker_size = self.get_parameter('marker_size').value
        self.camera_matrix = None
        self.dist_coeffs = None

        self.pose_pub = self.create_publisher(
            PoseStamped, '/target_pose', 10
        )

        self.image_sub = self.create_subscription(
            Image, '/camera/image_raw', self.image_callback, 10
        )
        self.info_sub = self.create_subscription(
            CameraInfo, '/camera/camera_info', self.info_callback, 10
        )

    def info_callback(self, msg):
        if self.camera_matrix is None:
            self.camera_matrix = np.array(msg.k).reshape(3, 3)
            self.dist_coeffs = np.array(msg.d)

    def image_callback(self, msg):
        if self.camera_matrix is None:
            return

        frame = self.bridge.imgmsg_to_cv2(msg, 'bgr8')
        gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
        corners, ids, _ = self.detector.detectMarkers(gray)

        if ids is not None:
            rvecs, tvecs, _ = aruco.estimatePoseSingleMarkers(
                corners, self.marker_size,
                self.camera_matrix, self.dist_coeffs
            )

            for i in range(len(ids)):
                pose_cam = PoseStamped()
                pose_cam.header.stamp = self.get_clock().now().to_msg()
                pose_cam.header.frame_id = 'camera_color_optical_frame'
                pose_cam.pose.position.x = float(tvecs[i][0][0])
                pose_cam.pose.position.y = float(tvecs[i][0][1])
                pose_cam.pose.position.z = float(tvecs[i][0][2])

                rmat, _ = cv2.Rodrigues(rvecs[i])
                from tf_transformations import quaternion_from_matrix
                q = quaternion_from_matrix(
                    np.vstack([np.hstack([rmat, [[0],[0],[0]]]),
                              [0, 0, 0, 1]])
                )
                pose_cam.pose.orientation.x = q[0]
                pose_cam.pose.orientation.y = q[1]
                pose_cam.pose.orientation.z = q[2]
                pose_cam.pose.orientation.w = q[3]

                try:
                    trans = self.tf_buffer.lookup_transform(
                        'base_link', 'camera_color_optical_frame',
                        rclpy.time.Time()
                    )
                    pose_base = do_transform_pose(pose_cam.pose, trans)
                    result = PoseStamped()
                    result.header.stamp = self.get_clock().now().to_msg()
                    result.header.frame_id = 'base_link'
                    result.pose = pose_base
                    self.pose_pub.publish(result)
                except Exception as e:
                    pass

        cv2.imshow('AR Localization', frame)
        cv2.waitKey(1)


def main(args=None):
    rclpy.init(args=args)
    node = ProjectARLocalizer()
    try:
        rclpy.spin(node)
    except KeyboardInterrupt:
        cv2.destroyAllWindows()
    finally:
        node.destroy_node()
        rclpy.shutdown()


if __name__ == '__main__':
    main()
```

### 35.3.3 模块3：MoveIt2抓取执行模块

```python
#!/usr/bin/env python3
"""模块3：MoveIt2抓取执行模块"""
import time
import rclpy
from rclpy.node import Node
from geometry_msgs.msg import PoseStamped
from moveit.planning import MoveItPy, PlanningComponent
from moveit_msgs.msg import CollisionObject, AttachedCollisionObject
from shape_msgs.msg import SolidPrimitive
from std_srvs.srv import Trigger

class ProjectGraspExecutor(Node):
    """综合实训 - 抓取执行模块"""
    def __init__(self):
        super().__init__('project_grasp_executor')
        self.moveit = MoveItPy(node_name='moveit_py')
        self.psm = self.moveit.get_planning_scene_monitor()
        self.arm = PlanningComponent(self.moveit, 'arm_group', 'link5')
        self.gripper = PlanningComponent(
            self.moveit, 'gripper', 'gripper_link'
        )

        self.arm.set_goal_position_tolerance(0.01)
        self.arm.set_goal_orientation_tolerance(0.02)
        self.arm.set_max_velocity_scaling_factor(0.4)

        # 设置场景
        self.setup_scene()

        # 订阅目标位姿
        self.pose_sub = self.create_subscription(
            PoseStamped, '/target_pose', self.target_callback, 10
        )

        # 服务接口：触发抓取
        self.grasp_srv = self.create_service(
            Trigger, '/grasp_execute', self.grasp_service_callback
        )

        self.current_target = None
        self.get_logger().info('抓取执行模块已启动')

    def setup_scene(self):
        """设置规划场景"""
        table = CollisionObject()
        table.id = 'table'
        table.header.frame_id = 'base_link'
        table.operation = CollisionObject.ADD
        prim = SolidPrimitive()
        prim.type = SolidPrimitive.BOX
        prim.dimensions = [0.8, 0.6, 0.02]
        table.primitives = [prim]
        pose = PoseStamped()
        pose.header.frame_id = 'base_link'
        pose.pose.position.z = -0.01
        pose.pose.orientation.w = 1.0
        table.primitive_poses = [pose.pose]
        self.psm.process_collision_object(table)

    def target_callback(self, msg):
        self.current_target = msg

    def grasp_service_callback(self, request, response):
        """服务回调：执行完整抓取流程"""
        if self.current_target is None:
            response.success = False
            response.message = '没有可用的目标位姿'
            return response

        success = self.execute_grasp_sequence(self.current_target)
        response.success = success
        response.message = '抓取成功' if success else '抓取失败'
        return response

    def execute_grasp_sequence(self, target):
        """完整抓取序列"""
        self.get_logger().info('=== 开始抓取序列 ===')

        # 1. 打开夹爪
        self.gripper.set_start_state_to_current_state()
        self.gripper.set_joint_value_target([0.65, 0.65])
        plan = self.gripper.plan()
        if plan:
            self.gripper.execute(plan.trajectory)
        time.sleep(0.3)

        # 2. 预抓取位姿
        pre_grasp = PoseStamped()
        pre_grasp.header = target.header
        pre_grasp.pose = target.pose
        pre_grasp.pose.position.z += 0.12

        self.arm.set_start_state_to_current_state()
        self.arm.set_pose_target(pre_grasp.pose, 'link5')
        plan = self.arm.plan()
        if not plan:
            self.get_logger().error('预抓取规划失败')
            return False
        self.arm.execute(plan.trajectory)
        time.sleep(0.5)

        # 3. 抓取位姿
        self.arm.set_start_state_to_current_state()
        self.arm.set_pose_target(target.pose, 'link5')
        plan = self.arm.plan()
        if not plan:
            self.get_logger().error('抓取规划失败')
            return False
        self.arm.execute(plan.trajectory)
        time.sleep(0.3)

        # 4. 闭合夹爪
        self.gripper.set_start_state_to_current_state()
        self.gripper.set_joint_value_target([0.0, 0.0])
        plan = self.gripper.plan()
        if plan:
            self.gripper.execute(plan.trajectory)
        time.sleep(0.5)

        # 5. 附着物体
        self.attach_object('workpiece')
        time.sleep(0.3)

        # 6. 撤离
        retreat = PoseStamped()
        retreat.header = target.header
        retreat.pose = target.pose
        retreat.pose.position.z += 0.15

        self.arm.set_start_state_to_current_state()
        self.arm.set_pose_target(retreat.pose, 'link5')
        plan = self.arm.plan()
        if not plan:
            self.get_logger().error('撤离规划失败')
            return False
        self.arm.execute(plan.trajectory)

        self.get_logger().info('=== 抓取完成 ===')
        return True

    def attach_object(self, object_id):
        """附着物体"""
        co = CollisionObject()
        co.id = object_id
        co.header.frame_id = 'gripper_link'
        co.operation = CollisionObject.ADD
        prim = SolidPrimitive()
        prim.type = SolidPrimitive.BOX
        prim.dimensions = [0.04, 0.04, 0.08]
        pose = PoseStamped()
        pose.header.frame_id = 'gripper_link'
        pose.pose.position.z = 0.04
        pose.pose.orientation.w = 1.0
        co.primitives = [prim]
        co.primitive_poses = [pose.pose]
        aco = AttachedCollisionObject()
        aco.link_name = 'gripper_link'
        aco.object = co
        aco.touch_links = ['gripper_link', 'link5']
        self.psm.process_attached_collision_object(aco)

    def detach_object(self, object_id):
        """分离物体"""
        co = CollisionObject()
        co.id = object_id
        co.operation = CollisionObject.REMOVE
        aco = AttachedCollisionObject()
        aco.link_name = 'gripper_link'
        aco.object = co
        self.psm.process_attached_collision_object(aco)


def main(args=None):
    rclpy.init(args=args)
    node = ProjectGraspExecutor()
    rclpy.spin(node)
    node.destroy_node()
    rclpy.shutdown()


if __name__ == '__main__':
    main()
```

### 35.3.4 模块4：主控编排模块

```python
#!/usr/bin/env python3
"""模块4：主控编排 - 全流程协调"""
import rclpy
from rclpy.node import Node
from rclpy.action import ActionServer
from std_msgs.msg import String
from std_srvs.srv import Trigger
import json
import asyncio

class ProjectOrchestrator(Node):
    """综合实训 - 主控编排模块"""
    def __init__(self):
        super().__init__('project_orchestrator')

        # 任务状态
        self.task_queue = []
        self.is_busy = False

        # 服务客户端：触发抓取
        self.grasp_client = self.create_client(
            Trigger, '/grasp_execute'
        )
        while not self.grasp_client.wait_for_service(timeout_sec=1.0):
            self.get_logger().info('等待抓取服务上线...')

        # 订阅任务指令
        self.task_sub = self.create_subscription(
            String, '/project/task', self.task_callback, 10
        )

        # 发布状态
        self.status_pub = self.create_publisher(
            String, '/project/status', 10
        )

        self.get_logger().info('主控编排模块已启动')

    def task_callback(self, msg):
        """收到新任务"""
        self.get_logger().info(f'收到任务: {msg.data}')

        try:
            task = json.loads(msg.data)
            self.task_queue.append(task)
            if not self.is_busy:
                self.execute_next_task()
        except json.JSONDecodeError:
            self.get_logger().error('任务格式错误')

    def execute_next_task(self):
        """执行下一个任务"""
        if not self.task_queue:
            self.is_busy = False
            return

        self.is_busy = True
        task = self.task_queue.pop(0)
        task_type = task.get('type', 'grasp')

        self.publish_status(f'执行任务: {task_type}')

        if task_type == 'grasp':
            self.execute_grasp_task(task)
        elif task_type == 'pick_and_place':
            self.execute_pick_and_place(task)
        elif task_type == 'batch_sort':
            self.execute_batch_sort(task)

    def execute_grasp_task(self, task):
        """执行抓取任务"""
        self.publish_status('开始抓取...')

        req = Trigger.Request()
        future = self.grasp_client.call_async(req)
        future.add_done_callback(self.grasp_done_callback)

    def grasp_done_callback(self, future):
        """抓取完成回调"""
        try:
            response = future.result()
            if response.success:
                self.publish_status('抓取成功')
            else:
                self.publish_status(f'抓取失败: {response.message}')
        except Exception as e:
            self.publish_status(f'抓取异常: {e}')

        # 执行下一个任务
        self.execute_next_task()

    def execute_pick_and_place(self, task):
        """执行抓取+放置任务"""
        self.publish_status('执行抓取放置任务')
        # 1. 抓取
        req = Trigger.Request()
        future = self.grasp_client.call_async(req)
        future.add_done_callback(
            lambda f: self.place_after_grasp(f, task)
        )

    def place_after_grasp(self, grasp_future, task):
        """抓取后执行放置"""
        if grasp_future.result().success:
            self.publish_status('抓取完成，准备放置')
            # 放置逻辑（移动到指定位置，释放物体）
            self.publish_status('放置完成')
        else:
            self.publish_status('抓取失败，跳过放置')

        self.execute_next_task()

    def execute_batch_sort(self, task):
        """批量分拣任务"""
        objects = task.get('objects', [])
        self.publish_status(f'开始分拣 {len(objects)} 个物体')

        # 循环分拣
        for i, obj in enumerate(objects):
            self.publish_status(f'分拣 {i+1}/{len(objects)}: {obj}')
            req = Trigger.Request()
            future = self.grasp_client.call_async(req)
            # 等待抓取完成
            rclpy.spin_until_future_complete(self, future)
            if future.result().success:
                self.publish_status(f'{obj} 分拣完成')
            else:
                self.publish_status(f'{obj} 分拣失败')

        self.publish_status('批量分拣完成')
        self.execute_next_task()

    def publish_status(self, status_text):
        """发布状态信息"""
        msg = String()
        msg.data = status_text
        self.status_pub.publish(msg)
        self.get_logger().info(f'状态: {status_text}')


def main(args=None):
    rclpy.init(args=args)
    node = ProjectOrchestrator()
    rclpy.spin(node)
    node.destroy_node()
    rclpy.shutdown()


if __name__ == '__main__':
    main()
```

```python
class RobustGraspExecutor(ProjectGraspExecutor):
    """带重试机制的抓取执行器"""

    def execute_with_retry(self, target, max_retries=3):
        for attempt in range(max_retries):
            self.get_logger().info(f'抓取尝试 {attempt + 1}/{max_retries}')

            success = self.execute_grasp_sequence(target)
            if success:
                return True

            self.get_logger().warn(f'尝试 {attempt + 1} 失败')
            time.sleep(1.0)

            # 回到安全位姿
            self.arm.set_named_target('home')
            plan = self.arm.plan()
            if plan:
                self.arm.execute(plan.trajectory)

        self.get_logger().error('所有抓取尝试均失败')
        return False
```
